let currentTab = 'files'; let activeFilter = 'All'; let dbInstance = null; let allFiles = []; let foldersList = []; // Clean state: populated solely by Admin // 1. Unified Taxonomy Loader (supports both 'folders' and 'shoaib_folders') function loadTaxonomies() { const stored = localStorage.getItem('folders') || localStorage.getItem('shoaib_folders'); if (stored) { try { const parsed = JSON.parse(stored); // Handles both string arrays ["Geography"] and object arrays [{name: "Geography"}] foldersList = parsed.map(item => (typeof item === 'object' && item !== null) ? (item.name || item.folderName) : item); } catch (e) { console.error("Error reading folder data:", e); foldersList = []; } } else { foldersList = []; } } // 2. Real-time Cross-Tab Sync (Triggers when Admin adds or deletes a folder) window.addEventListener('storage', (event) => { if (event.key === 'folders' || event.key === 'shoaib_folders') { loadTaxonomies(); renderChips(); if (currentTab === 'folders') { renderFolders(); } else { renderFiles(); } } }); // 3. Open IndexedDB function initDB() { return new Promise((resolve, reject) => { const req = indexedDB.open("ShoaibPortalDB", 1); req.onupgradeneeded = (e) => { const db = e.target.result; if (!db.objectStoreNames.contains("files")) { db.createObjectStore("files", { keyPath: "id", autoIncrement: true }); } }; req.onsuccess = (e) => { dbInstance = e.target.result; resolve(dbInstance); }; req.onerror = (e) => reject(e); }); } // 4. Fetch uploaded files function fetchFiles() { return new Promise((resolve) => { if (!dbInstance) return resolve([]); const tx = dbInstance.transaction(["files"], "readonly"); const store = tx.objectStore("files"); const request = store.getAll(); request.onsuccess = () => resolve(request.result || []); request.onerror = () => resolve([]); }); } // 5. Tab Navigation Switch function switchTab(tab) { currentTab = tab; const navFiles = document.getElementById('navFiles'); const navFolders = document.getElementById('navFolders'); const filesContainer = document.getElementById('filesContainer'); const foldersContainer = document.getElementById('foldersContainer'); const chipsContainer = document.getElementById('chipsContainer'); const viewTitle = document.getElementById('viewTitle'); const viewSubtitle = document.getElementById('viewSubtitle'); if (tab === 'files') { navFiles.classList.add('active'); navFolders.classList.remove('active'); filesContainer.style.display = 'grid'; foldersContainer.style.display = 'none'; chipsContainer.style.display = 'flex'; viewTitle.textContent = "Latest Files"; viewSubtitle.textContent = "Verified classroom modules, notes & study materials"; renderFiles(); } else { navFolders.classList.add('active'); navFiles.classList.remove('active'); filesContainer.style.display = 'none'; foldersContainer.style.display = 'grid'; chipsContainer.style.display = 'none'; viewTitle.textContent = "Class Folders"; viewSubtitle.textContent = "Organized repository categories"; renderFolders(); } } // 6. Render Filter Chips function renderChips() { const container = document.getElementById('chipsContainer'); const categories = ['All', ...foldersList]; container.innerHTML = categories.map(cat => ` `).join(''); } function filterByChip(cat) { activeFilter = cat; renderChips(); renderFiles(); } // 7. Render Files function renderFiles() { const container = document.getElementById('filesContainer'); const search = (document.getElementById('searchInput').value || '').toLowerCase().trim(); const filtered = allFiles.filter(file => { const matchesCategory = (activeFilter === 'All') || (file.folder === activeFilter); const matchesSearch = file.name && file.name.toLowerCase().includes(search); return matchesCategory && matchesSearch; }); if (filtered.length === 0) { container.innerHTML = `
📄

No files found

Admin has not uploaded documents for this selection yet.

`; return; } container.innerHTML = filtered.map(file => { const sizeFormatted = file.size ? (file.size / 1024 < 1024 ? (file.size / 1024).toFixed(1) + ' KB' : (file.size / (1024 * 1024)).toFixed(1) + ' MB') : 'Unknown size'; const dateFormatted = file.date || 'Recent'; return `
📑
${file.name}
${file.folder || 'General'}
${sizeFormatted} ${dateFormatted}
`; }).join(''); } // 8. Render Folders (Reflects Admin folders dynamically) function renderFolders() { const container = document.getElementById('foldersContainer'); if (foldersList.length === 0) { container.innerHTML = `
📁

No folders created

Admin has not added any class folders yet.

`; return; } container.innerHTML = foldersList.map(folderName => { const count = allFiles.filter(f => f.folder === folderName).length; return `
📁
${folderName}
${count} ${count === 1 ? 'file' : 'files'}
`; }).join(''); } function openFolder(folderName) { activeFilter = folderName; switchTab('files'); } function handleSearch() { if (currentTab === 'files') { renderFiles(); } } // 9. Download File function downloadFile(fileId) { const file = allFiles.find(f => f.id === fileId); if (!file) return; const blob = file.blob instanceof Blob ? file.blob : new Blob([file.data || file.content || ""], { type: file.type || "application/octet-stream" }); const url = URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = file.name; document.body.appendChild(a); a.click(); document.body.removeChild(a); setTimeout(() => URL.revokeObjectURL(url), 1000); } // 10. Bootstrap async function boot() { loadTaxonomies(); renderChips(); try { await initDB(); allFiles = await fetchFiles(); } catch (err) { console.error("IndexedDB error:", err); } renderFiles(); } window.addEventListener('DOMContentLoaded', boot);
